def generate_ontology(tables):
    ontology = {'entities': [], 'relationships': []}
    table_data = {}
    for name, csv_content in tables.items():
        reader = csv.DictReader(io.StringIO(csv_content))
        cols = reader.fieldnames
        table_data[name] = cols
        ontology['entities'].append({'name': name, 'attributes': cols})
    
    # Infer relationships based on column name overlap (e.g., user_id)
    all_cols = {}
    for t, cols in table_data.items():
        for c in cols:
            if c.endswith('_id') and c != 'order_id': # Simple heuristic
                all_cols.setdefault(c, []).append(t)
    
    for col, tabs in all_cols.items():
        if len(tabs) > 1:
            ontology['relationships'].append({'from': tabs[0], 'to': tabs[1], 'key': col})
    return ontology
