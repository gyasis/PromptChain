import csv, json
from io import StringIO

def generate_ontology(tables):
    entities, entity_names = [], []
    for name, data in tables.items():
        reader = csv.DictReader(StringIO(data))
        attrs = list(reader.fieldnames) if reader.fieldnames else []
        ent_name = name.capitalize()
        entity_names.append(ent_name)
        entities.append({'type': ent_name, 'attributes': attrs})
    relationships = []
    for ent in entities:
        for attr in ent['attributes']:
            if attr.endswith('_id'):
                target = attr[:-3].capitalize()
                if target in entity_names:
                    relationships.append({'from': ent['type'], 'to': target, 'via': attr})
    return {'entities': entities, 'relationships': relationships}

if __name__ == "__main__":
    t1, t2 = 'id,name\n1,Alice', 'order_id,user_id\n101,1'
    tables = {'user': t1, 'order': t2}
    try:
        out = generate_ontology(tables)
        assert len(out['relationships']) > 0 and any(r['to'] == 'User' for r in out['relationships'])
        print("SUCCESS:")
    except Exception:
        print("FAILURE:")
