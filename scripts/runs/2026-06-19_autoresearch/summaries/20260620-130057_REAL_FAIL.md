import sqlite3, argparse
from faker import Faker
def main():
    p = argparse.ArgumentParser()
    p.add_argument('--rows', type=int, default=100000)
    p.add_argument('--db', default='customers.sqlite')
    a = p.parse_args()
    conn = sqlite3.connect(a.db)
    conn.execute('PRAGMA journal_mode=WAL')
    conn.execute('DROP TABLE IF EXISTS customers')
    conn.execute('CREATE TABLE customers (customer_id INTEGER PRIMARY KEY, name TEXT, email TEXT, signup_date TEXT, total_spent REAL)')
    fake = Faker()
    batch = []
    for i in range(1, a.rows + 1):
        spent = round(fake.random.uniform(0, 5000), 2) if fake.random.randint(0, 10) > 1 else None
        date = str(fake.date_between(start_date='-2y', end_date='today')) if fake.random.randint(0, 10) > 1 else None
        batch.append((i, fake.name(), fake.email(), date, spent))
        if len(batch) >= 1000:
            conn.executemany('INSERT INTO customers VALUES (?,?,?,?,?)', batch)
            conn.commit()
            batch = []
    if batch:
        conn.executemany('INSERT INTO customers VALUES (?,?,?,?,?)', batch)
        conn.commit()
    conn.close()
if __name__ == '__main__': main()
