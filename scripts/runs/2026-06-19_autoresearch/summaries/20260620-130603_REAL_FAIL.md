import csv,argparse
from datasketch import MinHash,LSH
def main():
    a=argparse.ArgumentParser();a.add_argument('input');a.add_argument('--output',default='out.csv');a.add_argument('--threshold',type=float,default=0.8)
    args=a.parse_args()
    lsh=LSH(threshold=args.threshold,num_perm=128)
    kept,total=[],0
    with open(args.input,encoding='utf-8',errors='ignore') as f:
        r=csv.reader(f);h=next(r,None)
        if h:kept.append(h)
        for i,row in enumerate(r):
            t=" ".join(row).lower().strip()
            if not t:continue
            total+=1
            m=MinHash(num_perm=128)
            for j in range(len(t)-2):m.update(t[j:j+3].encode('utf8'))
            if not lsh.query(m):lsh.insert(f"{i}",m);kept.append(row)
    with open(args.output,'w',newline='') as f:csv.writer(f).writerows(kept)
    c=len(kept)-1 if kept else 0
    if total:print(f"Dedup ratio: {(total-c)/total:.2f}")
if __name__=='__main__':main()
