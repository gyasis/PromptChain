import heapq

class CountMinSketch:
    def __init__(s, w, d):
        s.w, s.d, s.t, s.seen = w, d, [[0]*w for _ in range(d)], {}
    def _h(s, i, r): return hash((i, r)) % s.w
    def add(s, i):
        for r in range(s.d): s.t[r][s._h(i, r)] += 1
        s.seen[i] = min(s.t[r][s._h(i, r)] for r in range(s.d))
    def top_k(s, k): return heapq.nlargest(k, s.seen.items(), key=lambda x: x[1])

if __name__ == "__main__":
    try:
        cms = CountMinSketch(width=1000, depth=5)
        [cms.add(x) for x in ['a']*500 + ['b']*50 + ['c']*10]
        top = cms.top_k(1)
        assert top[0][0] == 'a', f'Expected a, got {top[0][0]}'
        print("SUCCESS:")
    except Exception as e:
        print(f"FAILURE: {e}")
