import time
class TokenBucket:
    def __init__(s,c,r):s.c,s.r,s.t,s.l=c,r,c,time.time()
    def consume(s):
        n=time.time();s.t=min(s.c,s.t+(n-s.l)*s.r);s.l=n
        if s.t>=1:s.t-=1;return True
        return False
if __name__=="__main__":
    b=TokenBucket(5,1);r=[b.consume() for _ in range(6)]
    print("SUCCESS:"if r==[True]*5+[False]else"FAILURE:")
