```python
import pandas as pd; from thefuzz import fuzz
def fuzzy_join(t1,t2,c):
    if t1.empty or t2.empty: return pd.DataFrame(),0.0
    m=pd.merge(t1,t2,how='cross',suffixes=('_1','_2'))
    m['s']=m.apply(lambda r:fuzz.token_set_ratio(r[c+'_1'],r[c+'_2']),axis=1)
    res=m[m['s']>=80].drop_duplicates(subset=[c+'_1'])
    return res,len(res)/len(t1)
```
