## Method (cited)
Implement the P2 algorithm (Jain & Chlamtac) for dynamic quantile estimation to compute the streaming median. This method maintains 5 markers (minimum, lower, median, upper, maximum) and their desired positions to adjust quantile estimates dynamically without storing the full history [web:https://stackoverflow.com/questions/10657503/find-running-median-from-a-stream-of-integers]. Unlike windowed approaches like `RollingQuantile` which store a fixed subset of data [corpus:31ad179a-4cc8-11ef-98fa-fcb3bcabaf0e], P2 maintains constant memory O(1) regardless of stream length. The implementation will initialize markers after observing the first 5 data points, then update marker heights and positions for each subsequent value.

class P2Estimator:
    def __init__(s):
        s.c,s.b,s.q,s.n,s.np,s.dn=0,[],[0.]*5,[0]*5,[0.]*5,[0.,0.25,0.5,0.75,1.]
        s.i=False
    def update(s,x):
        s.c+=1
        if not s.i:
            s.b.append(x)
            if len(s.b)==5:
                s.b.sort();s.q=s.b[:];s.n=[1,2,3,4,5];s.np=[1.,2.,3.,4.,5.];s.i=True
            return
        k=0
        if x<s.q[0]:s.q[0]=x;k=0
        elif x>=s.q[4]:s.q[4]=x;k=3
        else:
            for j in range(1,5):
                if x<s.q[j]:k=j-1;break
        for j in range(k+1,5):s.n[j]+=1
        for j in range(5):s.np[j]+=s.dn[j]
        for j in range(1,4):
            d=s.np[j]-s.n[j]
            if (d>=1 and s.n[j+1]-s.n[j]>1)or(d<=-1 and s.n[j-1]-s.n[j]<-1):
                ds=1 if d>0 else -1
                qn=s.para(j,ds)
                s.q[j]=qn if s.q[j-1]<qn<s.q[j+1]else s.lin(j,ds)
                s.n[j]+=ds
    def para(s,j,d):
        qi,qi1,qi2=s.q[j],s.q[j-1],s.q[j+1]
        ni,ni1,ni2=s.n[j],s.n[j-1],s.n[j+1]
        t1=d/(ni2-ni1)
        t2=(ni-ni1+d)*(qi2-qi)/(ni2-ni)
        t3=(ni2-ni-d)*(qi-qi1)/(ni-ni1)
        return qi+t1*(t2+t3)
    def lin(s,j,d):
        qi=s.q[j]
        return qi+d*(s.q[j+1]-qi)/(s.n[j+1]-s.n[j])if d>0 else qi+d*(qi-s.q[j-1])/(s.n[j]-s.n[j-1])
    def median(s):
        if not s.i:
            b=sorted(s.b);n=len(b)
            return 0 if n==0 else b[n//2]if n%2==1 else (b[n//2-1]+b[n//2])/2
        return s.q[2]

if __name__=="__main__":
