import hashlib, math

class BloomFilter:
    def __init__(s, n, p):
        s.m = int(-n * math.log(p) / math.log(2)**2)
        s.k = int(s.m / n * math.log(2))
        s.ba = bytearray((s.m >> 3) + 1)
    def _h(s, k, i):
        h1 = int(hashlib.md5(k).hexdigest(), 16)
        h2 = int(hashlib.sha1(k).hexdigest(), 16)
        return (h1 + i * h2) % s.m
    def add(s, item):
        k = str(item).encode()
        for i in range(s.k):
            idx = s._h(k, i)
            s.ba[idx >> 3] |= 1 << (idx & 7)
    def check(s, item):
        k = str(item).encode()
        for i in range(s.k):
            idx = s._h(k, i)
            if not (s.ba[idx >> 3] & 1 << (idx & 7)): return False
        return True

if __name__ == "__main__":
    n, p = 1000, 0.01
    bf = BloomFilter(n, p)
    for i in range(n): bf.add(i)
    fps = sum(bf.check(i) for i in range(n, n * 3))
    print("SUCCESS:" if fps / (n * 2) <= p else "FAILURE:")
