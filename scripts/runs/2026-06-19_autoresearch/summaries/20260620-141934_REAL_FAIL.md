# dq_scan.py
import pandas as pd,json,sys
def main():
    try:df=pd.read_csv(sys.argv[1],encoding='utf-8')
    except UnicodeDecodeError:df=pd.read_csv(sys.argv[1],encoding='latin-1')
    if df.empty:print('EMPTY_FILE');sys.exit(1)
    r={'columns':{}}
    for c in df.columns:
        col=df[c];n=int(col.isnull().sum());o=0;d=False
        if col.dtype.kind in'biufc':
            m,s=col.mean(),col.std()
            if s>0:o=int((col>m+3*s).sum())
        elif col.dtype=='object'and col.map(type).nunique()>1:d=True
        r['columns'][c]={'nulls':n,'outliers':o,'drift':d}
    print(json.dumps(r))
if __name__=='__main__':main()
