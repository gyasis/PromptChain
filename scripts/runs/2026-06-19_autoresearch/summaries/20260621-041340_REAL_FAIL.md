```python
import csv, json, io

def csv_stats(s: str) -> str:
    a = {}
    for r in csv.DictReader(io.StringIO(s)):
        for c, v in r.items():
            if c not in a: a[c] = {"c":0,"s":0,"mi":None,"ma":None,"n":True}
            d = a[c]
            if not d["n"] or not v: continue
            try:
                n = int(v) if "." not in v else float(v)
            except:
                d["n"] = False
                continue
            d["c"] += 1; d["s"] += n
            d["mi"] = n if d["mi"] is None else min(d["mi"], n)
            d["ma"] = n if d["ma"] is None else max(d["ma"], n)
    return json.dumps({c: {"min": d["mi"], "max": d["ma"], "mean": d["s"]/d["c"]} 
                        for c, d in a.items() if d["n"] and d["c"]},
                      sort_keys=True)
```
