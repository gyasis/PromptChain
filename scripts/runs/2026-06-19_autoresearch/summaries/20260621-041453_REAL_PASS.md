def flatten(d, p=''):
    r = {}
    for k, v in d.items():
        n = f"{p}.{k}" if p else k
        r.update(flatten(v, n)) if isinstance(v, dict) else r.__setitem__(n, v)
    return r

def unflatten(d):
    r = {}
    for k, v in d.items():
        cur = r
        *heads, tail = k.split('.')
        for h in heads:
            cur = cur.setdefault(h, {})
        cur[tail] = v
    return r

orig = {
    "a": {"b": {"c": 1, "d": 2}, "e": []},
    "f": {"g": {"h": {"i": "x"}}},
    "j": 3,
}
flat = flatten(orig)
restored = unflatten(flat)
print("SUCCESS" if orig == restored else "FAILURE")
