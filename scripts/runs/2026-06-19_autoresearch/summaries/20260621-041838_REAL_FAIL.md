/// Encode a string with Run‑Length Encoding.
/// Consecutive identical characters become <char><count>.
pub fn rle_encode(data: &str) -> String {
    let mut encoded = String::new();
    let mut chars = data.chars();
    if let Some(mut prev) = chars.next() {
        let mut count = 1usize;
        for c in chars {
            if c == prev {
                count += 1;
            } else {
                encoded.push(prev);
                encoded.push_str(&count.to_string());
                prev = c;
                count = 1;
            }
        }
        // tail
        encoded.push(prev);
        encoded.push_str(&count.to_string());
    }
    encoded
}
