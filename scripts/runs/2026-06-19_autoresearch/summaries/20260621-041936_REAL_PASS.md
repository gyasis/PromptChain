The algorithm is a single linear pass over the input list:

1. Initialise `totals = defaultdict(int)`.

from collections import defaultdict
def group_sum(p):
    d=defaultdict(int)
    for k,v in p:
        if not isinstance(v,(int,float)): raise TypeError
        d[k]+=v
    return dict(d)
