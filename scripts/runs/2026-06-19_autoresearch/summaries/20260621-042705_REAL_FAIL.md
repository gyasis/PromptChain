```python
# csv_json.py
import csv, json, io

def csv_to_json(csv_path, json_path):
    with open(csv_path, newline='') as f:
        r = csv.DictReader(f)
        rows = list(r)
        hdr = r.fieldnames
    with open(json_path, 'w', encoding='utf-8') as f:
        json.dump(rows, f, separators=(',', ':'))
    return hdr

def roundtrip_check(csv_path, json_path):
    hdr = csv_to_json(csv_path, json_path)
    with open(json_path) as f:
        data = json.load(f)
    out = io.StringIO()
    w = csv.DictWriter(out, fieldnames=hdr, lineterminator='\n')
    w.writeheader()
    w.writerows(data)
    regenerated = out.getvalue()
    original = open(csv_path, newline='').read()
    return regenerated == original
```
