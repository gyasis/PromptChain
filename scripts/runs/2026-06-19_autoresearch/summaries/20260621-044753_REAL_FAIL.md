# bloom.py
import hashlib, math

class Bloom:
    def __init__(self, m, k):
        self.m, self.k = m, k
        self.b = bytearray((m + 7) // 8)

    def _idx(self, x):
        for i in range(self.k):
            h = hashlib.sha256(x.encode() + i.to_bytes(4, "little")).digest()
            yield int.from_bytes(h[:4], "big") % self.m

    def add(self, x):
        for i in self._idx(x):
            self.b[i >> 3] |= 1 << (i & 7)

    def __contains__(self, x):
        return all(self.b[i >> 3] & (1 << (i & 7)) for i in self._idx(x))
