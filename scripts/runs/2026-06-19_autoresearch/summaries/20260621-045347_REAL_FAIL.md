**Overview**  
We will implement a reusable **Python‑only fuzzy‑join** that:  

1. Loads two CSV tables (`left.csv` and `right.csv`) into **pandas.DataFrame** objects.  
2. Performs cheap *blocking* on the first character of the name column to keep the candidate‑pair space tractable.

# fuzzy_join.py
import sys, pandas as pd
from rapidfuzz import fuzz

LEFT, RIGHT = "name_left", "name_right"
THRESH = 80

df_l = pd.read_csv("left.csv")
df_r = pd.read_csv("right.csv")
df_l["blk"] = df_l[LEFT].str[0].str.lower()
df_r["blk"] = df_r[RIGHT].str[0].str.lower()

matched = set()
for blk, lgrp in df_l.groupby("blk"):
    rgrp = df_r[df_r["blk"] == blk]
    if rgrp.empty: continue
    for i, lrow in lgrp.iterrows():
        best = -1
        for _, rrow in rgrp.iterrows():
            s = fuzz.token_set_ratio(lrow[LEFT], rrow[RIGHT])
            if s >= THRESH and s > best:
                best = s
        if best > -1: matched.add(i)

rate = len(matched) / len(df_l)
print(f"Match rate: {rate:.4f}")
