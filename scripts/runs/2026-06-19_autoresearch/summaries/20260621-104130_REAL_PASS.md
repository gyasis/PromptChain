    if not s:
        return ""
    out = []
    prev = s[0]
    count = 1
    for ch in s[1:]:

        c = s[i]; cnt = 1
        while i + cnt < len(s) and s[i + cnt] == c:

    out = ""; i = 0

        c = t[i]; i += 1
        j = i
