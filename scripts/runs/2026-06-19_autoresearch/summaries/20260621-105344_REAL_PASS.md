import csv, json, io, sys
data = "name,age\nAlice,30\nBob,25\n"
inp, out = io.StringIO(data), io.StringIO()
for r in csv.DictReader(inp):
    out.write(json.dumps(r, separators=(',', ':')) + "\n")
exp = '{"name":"Alice","age":"30"}\n{"name":"Bob","age":"25"}\n'
if out.getvalue() == exp:
    print("SUCCESS:")
else:
    print("FAILURE:")
