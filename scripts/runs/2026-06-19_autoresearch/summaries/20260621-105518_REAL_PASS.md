import time, sys

class TokenBucket:
    def __init__(self, cap):
        self.cap = self.tok = self.rate = float(cap)
        self.last = time.monotonic()
    def consume(self, n=1):
        now = time.monotonic()
        self.tok = min(self.cap, self.tok + (now - self.last) * self.rate)
        self.last = now
        if self.tok >= n:
            self.tok -= n
            return True
        return False

def _test():
    N = 5
    b = TokenBucket(N)

    # first burst: N+2 attempts, only N should succeed
    ok1 = sum(b.consume() for _ in range(N + 2)) == N

    # wait >1 s so the bucket refills, then another N attempts – all should succeed
    time.sleep(1.1)
    ok2 = sum(b.consume() for _ in range(N)) == N

    if ok1 and ok2:
        print("SUCCESS: token bucket enforces", N, "req/s")
    else:
        print("FAILURE: token bucket broken")
        sys.exit(1)

if __name__ == "__main__":
    _test()
