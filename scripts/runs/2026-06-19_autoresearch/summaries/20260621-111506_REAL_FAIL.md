2. **Choose the register count `m`** – The relative standard error of HyperLogLog is ≈ `1.04/√m`. To guarantee < 5 % error we need  

   \[
   \frac{1.04}{\sqrt{m}} \le 0.05 \;\Longrightarrow\; m \ge \left(\frac{1.04}{0.05}\right)^2 \approx 433.

```python
import sys, random, hashlib, math

class HLL:
    def __init__(self, p=9):                     # 2^p registers (p=9 → 512)
        self.p, self.m = p, 1 << p
        self.reg = [0] * self.m
        self.alpha = (0.673, 0.697, 0.709)[(self.m).bit_length() - 5] if self.m in (16, 32, 64) else 0.7213 / (1 + 1.079 / self.m)

    def add(self, x):
        h = int.from_bytes(hashlib.sha1(str(x).encode()).digest()[:8], 'big')      # 64‑bit hash
        idx = h >> (64 - self.p)                                                   # register index
        w = (h << self.p) & ((1 << 64) - 1)                                        # remaining bits
        bits = 64 - self.p
        if w == 0:
            r = bits + 1
        else:
            r = (bits - w.bit_length()) + 1                                        # rank
        if r > self.reg[idx]:
            self.reg[idx] = r

    def count(self):
        Z = sum(2.0 ** -v for v in self.reg)
        E = self.alpha * self.m * self.m / Z
        if E <= 2.5 * self.m:               # small‑range correction
            V = self.reg.count(0)
            if V:
                return int(self.m * math.log(self.m / V))
        return int(E)

def run(N=10000, p=9, repeats=5):
    for i in range(repeats):
        h = HLL(p)
        data = list(range(N))
        random.shuffle(data)
        for v in data:
            h.add(v)
        est = h.count()
        err = abs(est - N) / N
        print(f'run {i+1}: estimate={est} error={err:.2%}')
        if err > 0.05:
            print('FAILURE: error >5%')
            sys.exit(1)
    print('SUCCESS: all runs within 5%')
    sys.exit(0)

if __name__ == '__main__':
    run()
