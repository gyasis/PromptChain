```python
import os, json, sys
import openai

def label_reviews(reviews):
    prompt = """You are a sentiment classifier. Assign the label **positive** or **negative** to each review.

Examples:
Review: "excellent" → Label: positive
Review: "terrible" → Label: negative

Now classify the following reviews (output a JSON list of labels in the same order):
"""
    prompt += "\n".join(f'Review: "{r}"' for r in reviews) + "\nLabels:"

    response = openai.ChatCompletion.create(
        model="gpt-3.5-turbo",
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
    )
    text = response.choices[0].message.content.strip()
    # Expected format: ["positive", "negative", "positive"] (or negative for "ok")
    try:
        labels = json.loads(text)
    except json.JSONDecodeError:
        # fallback: split by lines and strip
        labels = [line.split(":")[-1].strip() for line in text.splitlines() if line]
    if len(labels) != len(reviews):
        raise ValueError("Not all reviews received a label")
    return labels
```

import sys, json, urllib.request, urllib.error

# ---- Configuration ----
ENDPOINT = "http://house-gpu.local:8773/v1/chat/completions"
MODEL_ID = "<HOUSE_MODELS_DIR>/mlx-llm/Qwen2.5-0.5B-Instruct-4bit"
REVIEWS = ["great", "awful", "ok"]          # inline sample data

# ---- Prompt (few‑shot, explicit handling for "ok") ----
prompt = """You are a sentiment classifier. Assign the label "positive" or "negative" to each review.
Examples:
Review: "excellent" → Label: positive
Review: "terrible" → Label: negative
Now classify the following reviews. Respond ONLY with a JSON list of labels in the same order, using "positive" for ambiguous cases like "ok".
"""
for r in REVIEWS:
    prompt += f'Review: "{r}"\n'
prompt += "Labels:"

payload = {
    "model": MODEL_ID,
    "messages": [{"role": "user", "content": prompt}],
    "temperature": 0,
    "max_tokens": 100,
}
data = json.dumps(payload).encode()

request = urllib.request.Request(
    ENDPOINT,
    data=data,
    headers={"Content-Type": "application/json"},
    method="POST",
)

try:
    with urllib.request.urlopen(request, timeout=20) as resp:
        response_body = resp.read().decode()
    response_json = json.loads(response_body)
    content = response_json["choices"][0]["message"]["content"]
    # Expect the model to return a JSON list like ["positive","negative","positive"]
    labels = json.loads(content.strip())
except Exception as e:
    print(f"FAILURE: {e}")
    sys.exit(1)

# ---- Validation ----
valid = (
    isinstance(labels, list) and
    len(labels) == len(REVIEWS) and
    all(l in ("positive", "negative") for l in labels)
)

if valid:
    print("SUCCESS:", labels)
    sys.exit(0)
else:
    print("FAILURE: invalid labels", labels)
    sys.exit(1)
